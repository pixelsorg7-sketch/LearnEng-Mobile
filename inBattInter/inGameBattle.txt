        const {showResults,setshowResults}=useContext(MyInGameStor);

        //animation ewan

          // State for the countdown number
  const [countdown, setCountdown] = useState(3); // Start from 3
  // State to control the visibility of the main content view
  const [showMainContent, setShowMainContent] = useState(false);

  // Animated value for the current countdown number's animation (scale and opacity)
  const numberAnim = useRef(new Animated.Value(0)).current; // 0 for initial scale/opacity

        useEffect(() => {
    // This effect runs whenever the 'countdown' state changes
    if (countdown > 0) {
      // Reset animation value for the new number
      numberAnim.setValue(0);

      // Sequence of animations for each number:
      // 1. Scale up and fade in (appear)
      // 2. Wait for a moment
      // 3. Scale down and fade out (disappear)
      Animated.sequence([
        Animated.parallel([
          Animated.timing(numberAnim, {
            toValue: 1, // Scale to 1 (normal size) and opacity to 1 (fully visible)
            duration: 400, // Fast appearance
            easing: Easing.easeOutBack, // A nice bouncy easing for scale
            useNativeDriver: true,
          }),
          Animated.timing(numberAnim, {
            toValue: 1, // Opacity to 1
            duration: 200, // Faster opacity fade in
            useNativeDriver: true,
          }),
        ]),
        Animated.delay(500), // Hold the number for a bit
        Animated.parallel([
          Animated.timing(numberAnim, {
            toValue: 0, // Scale back to 0 (disappear)
            duration: 300, // Fast disappearance
            easing: Easing.easeInBack, // Bouncy easing for scale out
            useNativeDriver: true,
          }),
          Animated.timing(numberAnim, {
            toValue: 0, // Opacity back to 0
            duration: 300, // Faster opacity fade out
            useNativeDriver: true,
          }),
        ]),
      ]).start(() => {
        // After the current number's animation sequence completes,
        // decrement the countdown. This triggers the effect again for the next number.
        setCountdown(prevCount => prevCount - 1);
      });
    } else if (countdown === 0) {
      // Once countdown reaches 0, fade in the main content
      setShowMainContent(true);
    }
  }, [countdown, numberAnim]); // Depend on countdown to re-trigger the animation logic

  // Animated style for the countdown number
  const countdownNumberStyle = {
    opacity: numberAnim.interpolate({
      inputRange: [0, 0.5, 1], // Opacity points
      outputRange: [0, 0.5, 1], // Values corresponding to inputRange
      extrapolate: 'clamp',
    }),
    transform: [
      {
        scale: numberAnim.interpolate({
          inputRange: [0, 0.5, 1],
          outputRange: [0.5, 1.2, 1], // Scale from smaller, slightly overshoot, then settle
          extrapolate: 'clamp',
        }),
      },
    ],
  };

  // Animated value for the main content's fade-in
  const mainContentOpacity = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    if (showMainContent) {
      Animated.timing(mainContentOpacity, {
        toValue: 1,
        duration: 800, // Smooth fade-in duration
        useNativeDriver: true,
        easing: Easing.ease,
      }).start();
    }
  }, [showMainContent, mainContentOpacity]);

  if (!showMainContent) {
    // Render the countdown animation
    return (
     <View style={styles.countdownContainer}>
<View style={styles.countdownsubContainer}>
        {countdown > 0 ? (
          <Animated.Text style={[styles.countdownText, countdownNumberStyle]}>
            {countdown}
          </Animated.Text>
        ) : (
          // You could show "Go!" or nothing here briefly
          <Text style={[styles.countdownText,{fontSize:50}]}>LETS LEARN</Text> // Or another animation
        )}
</View>
      </View>
    );
  }
else{
    return(

        <>
    
        
        <View style = {styles.container}>
 
        <ImageBackground
        style = {styles.bgcon}
        source={{uri:'https://img.freepik.com/free-vector/school-building-scene_1308-29938.jpg?ga=GA1.1.1339217056.1740616814&semt=ais_hybrid&w=740'}}
        >

        { showResults ? <ResultPanel/> : ''}  
       
        <View style = {styles.progbar}><ProgressBar/></View>

        <QuestionBox/>



      <View style = {styles.playercon}>
        <PlayerObj/>
        <EnemyObj/>
      </View>

       
        <View style = {styles.floor}>  

         <View style = {styles.upfloorrepo}><UpperFloor/></View>
       
         <View style = {styles.downfloorrepo}>
         <LowerFloor/>
     
         <View style = {styles.answercon}>
      
        <View style = {styles.anschoice}> <AnswerChoice/> </View> 
        </View>

         </View>

         </View>





        </ImageBackground>

       

        </View>

        </>
    );

}};
